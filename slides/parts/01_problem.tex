      \begin{itemize}
        \item Prices sit on a grid with tick size $\delta$ (currency).
        \item \textbf{Limit order}: rests in a first-in, first-out (FIFO) queue at a price.
        \item \textbf{Market order}: trades at once against the best opposite queue.
        \item \textbf{Cancellation}: removes a resting order.
        \item Best bid $p_B$, best ask $p_A$, mid $S = (p_A + p_B)/2$, spread $s = (p_A - p_B)/\delta$ ticks, best queue sizes $q_b, q_a$ (orders).
      \end{itemize}
      \vspace{0.3em}

\begin{frame}{A liquidity provider's P\&L, exactly}
  Let fill $j$ happen at time $t_j$ and price $p_j$, with $\varepsilon_j = +1$ if the LP sold and $-1$ if it bought. Inventory is $y_t = -\sum_{t_j < t}\varepsilon_j$ (shares). Starting flat, the marked-to-market P\&L $\Pi_T$ (currency) is
  \[
    \Pi_T \;=\; \underbrace{\sum_j \varepsilon_j\,(p_j - S_{t_j})}_{\text{spread captured}\;\ge 0}
    \;+\; \underbrace{\int_0^T y_{t^-}\,\dd S_t}_{\text{inventory P\&L}} .
  \]
  This is an identity, not a model. Fill by fill, at a horizon $h$ (s) after the fill:
  \[
    \varepsilon_j\,(p_j - S_{t_j + h}) = \underbrace{\varepsilon_j\,(p_j - S_{t_j})}_{\text{half-spread}} - \underbrace{\varepsilon_j\,(S_{t_j+h} - S_{t_j})}_{\text{adverse selection}} .
  \]
  The left side is the \hl{markout} $M_j(h)$. LPs win the spread on every fill and lose it back when fills predict the next move. Every model in this deck is, for an LP, a model of one of these two terms.
\end{frame}

\begin{frame}{Facts the models must reproduce}
  \begin{enumerate}
    \small
    \item \textbf{Order signs have long memory.} The sign autocorrelation decays as a power law, $C_\varepsilon(\ell) \sim \ell^{-\gamma_\varepsilon}$ with $0 < \gamma_\varepsilon < 1$, over thousands of trades \citep{lillo2004long}.
    \item \textbf{Yet prices are close to unpredictable.} Impact must decay to offset the memory \citep{bouchaud2004fluctuations}.
    \item \textbf{Impact is concave in size.} A metaorder of $Q$ shares moves the price by about $Y\sigma_D\sqrt{Q/V_D}$, with daily volatility $\sigma_D$, daily volume $V_D$ and $Y$ of order one \citep{toth2011anomalous,donier2015fully}.
    \item \textbf{Order flow is close to critical.} Fitted Hawkes branching ratios are often near one \citep{hardiman2013critical}.
    \item \textbf{Imbalance predicts the next move.} Order flow imbalance explains a large share of short-horizon price changes \citep{cont2014price}, and deep models find the same price formation mechanism across stocks \citep{sirignano2019universal}.
    \item \textbf{Speed is worth money.} Latency races are frequent and a sizeable part of trading costs \citep{aquilina2022quantifying}.
  \end{enumerate}
\end{frame}
