\begin{frame}[fragile]{Everything here is reproducible}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      \textbf{Package} \texttt{stochastic\_lob}, NumPy and SciPy only:
      \begin{itemize}
        \item \texttt{markov\_lob}, \texttt{best\_queues}, \texttt{queue\_reactive}
        \item \texttt{hawkes}, \texttt{impact}, \texttt{latent\_book}, \texttt{microprice}
        \item \texttt{market\_making}, \texttt{adverse\_selection}, \texttt{queue\_position}, \texttt{markouts}, \texttt{amm}
        \item \texttt{lobster}: read and write LOBSTER order book files
      \end{itemize}
      \textbf{Tests} (46): each closed form or estimator against an independent computation (Monte Carlo, exact linear solve, PDE, or a known limit).

      \textbf{Experiments} (23): one script per figure; numbers go to \texttt{results/} as \LaTeX{} macros, and the slides read them.
    \end{column}
    \begin{column}{0.46\textwidth}
\begin{verbatim}
pip install -e .
python -m pytest tests
python experiments/run_all.py
python build_slides.py
\end{verbatim}
      \small
      Fixed random seeds; about 15 minutes on a laptop for all experiments.
    \end{column}
  \end{columns}
\end{frame}
